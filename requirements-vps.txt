# VPS / Linux runtime deps (extends requirements.txt)
#
# The Mac uses whisper-cli (whisper.cpp, Metal GPU) for transcription, so the
# base requirements.txt stays lean. On a headless Linux VPS there is no Metal
# and no whisper-cli, so social_downloader.py falls back to CPU transcription —
# faster-whisper (int8, RAM-light) is the preferred path.
#
# Install on the VPS:
#   pip install -r requirements.txt -r requirements-vps.txt
#
# System packages required first (apt):
#   sudo apt install -y ffmpeg
#   # playwright browser (only if using the browser-based video downloader):
#   python -m playwright install --with-deps chromium

-r requirements.txt

# CPU transcription (preferred fallback on Linux — see transcribe() in social_downloader.py)
faster-whisper>=1.0

# Video/audio extraction for the non-caption path. clip.py and scripts/brain_clip.py
# both `import yt_dlp` at runtime; clip_yt_channel.py uses it to list a channel's full
# history (falling back to the 15-video RSS feed when it is absent).
yt-dlp>=2024.1

# Browser-based video/audio capture (social_downloader.py).
# NOTE: pulls a full Chromium (~400MB disk + RAM during runs). On a small VPS
# consider switching the downloader to yt-dlp instead — lighter, no browser.
playwright>=1.40

# Document -> Markdown conversion (scripts/doc2md.py). Handles the formats the
# web/video paths don't: PDF, Word, PowerPoint, Excel, images. Local, no API cost.
markitdown[pdf,docx,pptx,xlsx]>=0.1

# Clean article extraction from URLs (scripts/doc2md.py article lane). trafilatura
# strips nav/ads/boilerplate -> just the article body. Local, no API key. Replaces
# the unconfigured `web_extract` tool for article/blog URLs.
trafilatura>=2.0

# JS-rendered/SPA article fallback. doc2md.py renders via playwright connect_over_cdp
# -> LightPanda (lightweight headless browser, ~10x lighter than Chromium), then
# re-extracts with trafilatura. LightPanda is NOT a pip package — install the binary
# on the VPS (free, open-source):
#   curl -fsSL -o /root/.local/bin/lightpanda \
#     https://github.com/lightpanda-io/browser/releases/download/nightly/lightpanda-x86_64-linux
#   chmod +x /root/.local/bin/lightpanda
# (the `playwright` pkg above is the CDP client; no `playwright install` needed to CDP-connect.)

# Semantic ("by meaning") search over the vault. scripts/semantic_index.py builds a
# local numpy embedding index (BAAI/bge-small, CPU, no API key); scripts/semantic_query.py
# searches it. obsidian_mcp.py's search_vault is hybrid (keyword + semantic). Refreshed
# nightly by the brain-semantic-reindex.timer (04:00 PT). numpy + onnxruntime are deps.
fastembed>=0.4
langchain-text-splitters  # book chunking, recursive separator ladder
